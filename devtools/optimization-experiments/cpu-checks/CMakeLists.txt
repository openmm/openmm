cmake_minimum_required(VERSION 3.18)
project(OpenMMExperimentCpuChecks LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(OPENMM_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(HilbertReference OBJECT "${OPENMM_SOURCE}/libraries/hilbert/src/hilbert.cpp")
target_include_directories(HilbertReference PRIVATE "${OPENMM_SOURCE}/libraries/hilbert/include")
target_compile_definitions(HilbertReference PRIVATE hilbert_c2i=referenceHilbertC2I)
add_executable(CheckHilbert check_hilbert.cpp $<TARGET_OBJECTS:HilbertReference>)
target_include_directories(CheckHilbert PRIVATE "${OPENMM_SOURCE}/libraries/hilbert/include" "${OPENMM_SOURCE}/platforms/common/src")
add_executable(CheckNeighborSkin check_neighbor_skin.cpp)
target_include_directories(CheckNeighborSkin PRIVATE "${OPENMM_SOURCE}/platforms/cuda/src")
add_test(NAME CheckHilbert COMMAND CheckHilbert)
add_test(NAME CheckNeighborSkin COMMAND CheckNeighborSkin)
if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
    set(SETTER_INCLUDES "${GENERATED}/exact_baseline.inc" "${GENERATED}/exact_candidate.inc" "${GENERATED}/exact_helper.inc" "${GENERATED}/exact_copy_coordinates.inc")
    add_custom_command(OUTPUT ${SETTER_INCLUDES}
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_setter.py" --source "${OPENMM_SOURCE}" --output "${GENERATED}"
        DEPENDS prepare_setter.py "${OPENMM_SOURCE}/platforms/common/src/CommonKernels.cpp" "${OPENMM_SOURCE}/platforms/common/src/kernels/copyCoordinateBuffers.cc"
        VERBATIM)
    add_executable(CheckSetter check_setter.cpp ${SETTER_INCLUDES})
    target_include_directories(CheckSetter PRIVATE "${GENERATED}" "${OPENMM_SOURCE}/openmmapi/include" "${OPENMM_SOURCE}/platforms/common/include")
    target_compile_options(CheckSetter PRIVATE /fp:precise)
    add_test(NAME CheckSetter COMMAND CheckSetter)
endif()
if(MSVC)
    target_compile_options(CheckNeighborSkin PRIVATE /fp:precise)
endif()
